          reference_id: 'FETCH_' + phone + '_' + Date.now(),
          generate_xml: false,
          generate_pdf: false,
        },
      },
      {
        headers: {
          'client_id': process.env.DECENTRO_CLIENT_ID,
          'client_secret': process.env.DECENTRO_CLIENT_SECRET,
          'Content-Type': 'application/json',
        },
      }
    );

    console.log('🔍 eAadhaar fetch result:', JSON.stringify(relayResponse.data));
    const d = relayResponse.data?.data;

    if (relayResponse.data?.status === 'SUCCESS' && d) {
      const aadhaarUid = d.aadhaarUid || '';
      const poi = d.proofOfIdentity || {};
      const poa = d.proofOfAddress || {};
      const address = [poa.house, poa.street, poa.landmark, poa.locality, poa.vtc, poa.postOffice, poa.subDistrict, poa.district, poa.state, poa.pincode].filter

      await User.findOneAndUpdate(
        { phone },
        {
          aadhaarVerified: true,
          aadhaarVerifiedAt: new Date(),
          aadhaarNumber: aadhaarUid || undefined,
          address: address || user.address,
          ...(poi.name ? { name: poi.name, panName: user.panName || poi.name } : {}),
          ...(poi.dob ? { dob: poi.dob } : {}),
          ...(poi.gender ? { gender: poi.gender } : {}),

      );

    res.json({ success: false, error: err.response?.data || err.message });
  }
});
